# Exact pins, not ranges -- and the SAME exact versions as
# ../agent-service/requirements.txt's scikit-learn/pandas/mlflow-skinny
# pins, on purpose. This is the fix for a real, observed bug: both files
# used to say `scikit-learn>=1.3,<2`, which let training (this file,
# resolved at whatever moment `run_training.sh` last ran) and serving
# (agent-service's image, resolved at whatever moment IT was last built)
# independently drift to different patch versions within that range --
# confirmed live as `InconsistentVersionWarning: Trying to unpickle
# estimator ... from version 1.7.0 when using version 1.9.0` in
# agent-service's logs. A loose range across a train/serve boundary is
# exactly the kind of version skew a real MLOps setup has to prevent;
# pinning both sides to the literal same version is the direct fix, not a
# cosmetic one. mlflow pinned to 3.15.1 to match the running server image
# (ghcr.io/mlflow/mlflow:latest resolved to 3.15.1 during verification) --
# a 2.x client against a 3.x server risks API drift (e.g. pyfunc
# model-loading internals changed between majors), found and fixed during
# this project's own testing.
mlflow==3.15.1
# 1.7.0, not a newer 1.x -- confirmed directly: run_training.sh's offline
# fallback path (pip download --platform manylinux2014_aarch64
# --python-version 311, used whenever container egress is broken) only has
# wheels published for scikit-learn up through 1.7.0 under that platform
# tag; 1.8.0/1.9.0 aren't available there. Pinning to a version installable
# via EITHER path (normal build or offline fallback) is what actually
# guarantees this doesn't drift again, not just picking whatever's newest.
scikit-learn==1.7.0
pandas==2.3.2
